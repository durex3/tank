## 一、原理

贴图不存最终颜色，只存形状、明度分层和侵蚀噪声：

| 通道 | 含义 | 取值 |
| --- | --- | --- |
| R | 卡通明度层级 | 0.30 暗带 / 0.70 主带 / 1.00 亮芯 |
| G | 侵蚀噪声 | 平滑灰度，用于生命后段破洞消散 |
| B | 未使用 | 0 |
| A | 覆盖度 | 主体 1，背景 0，边缘 1–2 像素抗锯齿 |

着色器拿 R 去查一条硬阶渐变，得到最终颜色；拿 G 与 life 比较，决定形状何时破洞。
所以生图时**只画灰度三阶形状**，颜色由项目渐变资源决定。

## 二、生图

### 输出位置

`assets/generated/vfx/toon-fx-v1/source/`

### 画布

1024×1024，2×2 等分四格，每格 512×512。主体位于各自格子中央 70%，格间留透明间隔。
真透明 RGBA。若只能出不透明图，出纯黑底并在文件名加 `-black`。

### 通用提示词（每次先贴）

```
Generate ONE 2x2 sprite sheet, 1024x1024, four equal quadrants, each effect centered in its
quadrant with transparent margins between cells. True transparent background (RGBA PNG).

Draw in GRAYSCALE ONLY using exactly three luminance levels: dark 30 percent gray, main
70 percent gray, hot core 100 percent white. Hard-edged flat shapes, no gradients inside a
shape. The three levels form the cel-shading bands of one cartoon effect.

HARD NEGATIVES:
NO color, NO outlines, NO strokes, NO borders, NO rim lines;
NO 3D extrusion, NO side faces, NO bevel, NO perspective;
NO glow, NO blur, NO soft shadow, NO drop shadow;
NO gradients inside shapes, NO noise, NO semi-transparent fringe;
NO text, NO labels, NO grid lines drawn in the image.

Style: Supercell-like cartoon combat VFX. Bold readable silhouettes, chunky asymmetric
shapes with intentional broken gaps, short directional bursts. Readable when each cell is
displayed at 48 pixels. All four cells must share one cohesive art style.
```

## 三、后处理脚本

把下面内容原样写入 `tools/prepare_toon_fx_131.py`。

```python
"""Split 2x2 grayscale sheets into R(luminance)/G(erosion)/A(coverage) textures.

Deterministic: same source and same seed always produce identical output.
G is generated here because image models cannot paint a data channel.
"""
from pathlib import Path
import json
import shutil
import subprocess

ROOT = Path(__file__).resolve().parents[1]
SRC = ROOT / "assets/generated/vfx/toon-fx-v1/source"
OUT = ROOT / "assets/generated/vfx/toon-fx-v1"
SIZE = 512
MAGICK = shutil.which("magick")
if not MAGICK:
    raise SystemExit("ImageMagick 7 (magick) is required")


def run(*args):
    return subprocess.check_output([MAGICK, *map(str, args)], text=True).strip()


def split(cell: Path, index: int, tag: str) -> Path:
    base = OUT / f"{tag}-{index}"
    r = base.with_name(base.name + "-r.png")
    a = base.with_name(base.name + "-a.png")
    g = base.with_name(base.name + "-g.png")
    b = base.with_name(base.name + "-b.png")
    # R: quantize luminance into exactly three cel bands.
    run(cell, "-colorspace", "Gray", "-resize", f"{SIZE}x{SIZE}!",
        "-fx", "(u<0.45?0.30:(u<0.85?0.70:1.0))", r)
    # A: transparency, with a little softness so edges do not alias.
    run(cell, "-alpha", "extract", "-resize", f"{SIZE}x{SIZE}!",
        "-level", "4%,20%", a)
    # G: deterministic erosion noise, one seed per cell.
    run("-size", f"{SIZE}x{SIZE}", "-seed", str(131 + index * 17 + len(tag)),
        "plasma:fractal", "-colorspace", "Gray", "-blur", "0x0.6", g)
    run("-size", f"{SIZE}x{SIZE}", "xc:black", b)
    run(r, g, b, a, "-combine", base.with_name(base.name + ".png"))
    for tmp in (r, a, g, b):
        tmp.unlink(missing_ok=True)
    return base.with_name(base.name + ".png")


sheets = sorted(SRC.glob("toon_*_2x2*.png"))
if not sheets:
    raise SystemExit(f"no source sheets in {SRC}")
report = []
for sheet in sheets:
    tag = sheet.stem.replace("_2x2", "").replace("-black", "")
    width, height = map(int, run("identify", "-format", "%w %h", sheet).split())
    if width != height:
        raise SystemExit(f"{sheet.name} is not square: {width}x{height}")
    half = width // 2
    cells = []
    for index, (x, y) in enumerate([(0, 0), (1, 0), (0, 1), (1, 1)]):
        cell = OUT / f".cell-{tag}-{index}.png"
        run(sheet, "-crop", f"{half}x{half}+{x * half}+{y * half}", "+repage", cell)
        cells.append(split(cell, index, tag))
        cell.unlink(missing_ok=True)
    atlas = OUT / f"{tag}_2x2.png"
    run("(", cells[0], cells[1], "+append", ")", "(", cells[2], cells[3], "+append", ")",
        "-append", atlas)
    for path in cells + [atlas]:
        report.append({
            "file": path.name,
            "image": run("identify", "-format", "%wx%h %[channels]", path),
            "alpha_range": run(path, "-alpha", "extract", "-format",
                               "%[fx:minima] %[fx:maxima]", "info:"),
            "luminance_range": run(path, "-colorspace", "Gray", "-format",
                                   "%[fx:minima] %[fx:maxima]", "info:"),
        })
(OUT / "normalization-report.json").write_text(json.dumps(report, indent=2) + "\n")
print(json.dumps(report, indent=2))
```

运行：

```bash
python3 tools/prepare_toon_fx_131.py
```

验收：报告中每张图 `alpha_range` 最小值为 0；`luminance_range` 最大值接近 1。

## 四、共享着色器

写入 `presentation/shaders/toon_fx.gdshader`。

```glsl
shader_type spatial;
render_mode unshaded, blend_mix, depth_draw_never, cull_disabled, shadows_disabled;
// R = cel band, G = erosion key, A = coverage. life is injected from committed ticks only.

uniform sampler2D sheet : filter_linear, repeat_disable;
uniform sampler2D ramp : source_color, filter_nearest, repeat_disable;
uniform vec2 frames = vec2(2.0, 2.0);
uniform float cell = 0.0;
uniform bool use_instance_cell = false;
uniform float erode_start = 0.35;
uniform float erode_softness = 0.06;
uniform float ramp_shift = 0.12;
uniform float life = 0.0;
uniform float camera_pull = 0.0;
uniform float spin = 0.0;
uniform float lift = 0.0;
uniform float opacity = 1.0;
uniform bool billboard = true;

void vertex() {
	if (billboard) {
		float angle = spin;
		VERTEX.xy = mat2(vec2(cos(angle), -sin(angle)), vec2(sin(angle), cos(angle))) * VERTEX.xy;
		VERTEX.y += lift;
		vec3 scale = vec3(length(MODEL_MATRIX[0].xyz), length(MODEL_MATRIX[1].xyz), length(MODEL_MATRIX[2].xyz));
		vec3 cam_z = normalize(INV_VIEW_MATRIX[2].xyz);
		MODELVIEW_MATRIX = VIEW_MATRIX * mat4(
			INV_VIEW_MATRIX[0] * scale.x,
			INV_VIEW_MATRIX[1] * scale.y,
			INV_VIEW_MATRIX[2] * scale.z,
			vec4(MODEL_MATRIX[3].xyz + cam_z * camera_pull, 1.0));
	}
	float index = use_instance_cell ? floor(clamp(INSTANCE_CUSTOM.z, 0.0, 0.999) * frames.x * frames.y) : cell;
	UV = (UV + vec2(mod(index, frames.x), floor(index / frames.x))) / frames;
}

void fragment() {
	vec4 texel = texture(sheet, UV);
	float value = clamp(texel.r - ramp_shift * life, 0.0, 1.0);
	vec3 colour = texture(ramp, vec2(value, 0.5)).rgb;
	float threshold = smoothstep(erode_start, 1.0, life);
	float keep = smoothstep(threshold, threshold + erode_softness, texel.g);
	ALBEDO = colour * COLOR.rgb;
	ALPHA = texel.a * keep * COLOR.a * opacity;
}
```

## 六、代码适配器

写入 `presentation/toon_fx_materials.gd`。

```gdscript
extends RefCounted
class_name ToonFxMaterials
## Adapter for the toon sheet shader. Life is injected from committed ticks only.

const MATERIALS: Dictionary = {
	&"warm": preload("res://content/presentation/toon_fx_warm.tres"),
	&"metal": preload("res://content/presentation/toon_fx_metal.tres"),
	&"water": preload("res://content/presentation/toon_fx_water.tres"),
	&"dust": preload("res://content/presentation/toon_fx_dust.tres"),
	&"guard": preload("res://content/presentation/toon_fx_guard.tres"),
}

static func apply(node: Sprite3D, sheet: Texture2D, family: StringName = &"warm",
		cell: int = 0, billboard: bool = true) -> void:
	var base: ShaderMaterial = MATERIALS.get(family, MATERIALS[&"warm"])
	var material: ShaderMaterial = base.duplicate()
	material.set_shader_parameter("sheet", sheet)
	material.set_shader_parameter("cell", float(cell))
	material.set_shader_parameter("billboard", billboard)
	node.material_override = material

static func advance(node: Sprite3D, life: float) -> void:
	if node.material_override is ShaderMaterial:
		(node.material_override as ShaderMaterial).set_shader_parameter("life", clampf(life, 0.0, 1.0))
```

每个实例必须 `duplicate()`，否则多张特效共享同一个 life，会同步消散。
